  const handleTopup = async (amount: number = 50) => {
    if (!userToken) return;
    setTopupLoading(true);
    try {
      const data = await userApi.topup(userToken, amount, handle401Error);
      Alert.alert('Top Up Success', `+${data.credits_added} credits\nTotal: ${data.credits_total} credits`);
      await refreshUserInfo();
    } catch (err: unknown) {
      Alert.alert('Top Up Failed', err instanceof Error ? err.message : 'Unknown error');
    } finally {
      setTopupLoading(false);
    }
  };

  const handleChangePassword = async () => {
    if (!newPassword || newPassword.length < 6) {
      Alert.alert('Error', 'New password must be at least 6 characters');
      return;
    }
    if (newPassword !== confirmPassword) {
      Alert.alert('Error', 'New password and confirmation do not match');
      return;
    }
    if (!userToken) return;
    setChangePwdLoading(true);
    try {
      await authApi.changePassword(userToken, oldPassword, newPassword, handle401Error);
      Alert.alert('Success', 'Password updated. Please login again.');
      setChangePwdVisible(false);
      setOldPassword('');
      setNewPassword('');
      setConfirmPassword('');
      await logout();
      router.replace('/auth/login');
    } catch (err: unknown) {
