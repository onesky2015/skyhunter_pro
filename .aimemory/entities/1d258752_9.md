export const formatDateTime = (isoStr: string | null): string => {
  if (!isoStr) return '--';
  try {
    const d = new Date(ensureUTC(isoStr));
    if (isNaN(d.getTime())) return '--';
    return d.toLocaleString(undefined, {
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
    });
  } catch {
    return '--';
  }
};
