  // 恢复监控任务
  const restoreMonitor = useCallback(async (id: number) => {
    try {
      await monitorApi.restoreMonitor(userToken, id, on401);
      setTrashMonitors((prev) => prev.filter((m) => m.id !== id));
      await fetchMonitors();
      Alert.alert('✅ Restored', `Monitor #${id} has been restored.`);
    } catch (err: unknown) {
      Alert.alert('Error', err instanceof Error ? err.message : 'Failed to restore monitor');
    }
  }, [userToken, on401, fetchMonitors]);

  // 更新监控任务状态
  const updateMonitorStatus = useCallback(async (id: number, status: string) => {
