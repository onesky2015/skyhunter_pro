import { useState, useCallback } from 'react';
import { Alert } from 'react-native';
import { MonitorTask, MonitorCreatePayload, MonitorUpdatePayload } from '../types';
import { monitorApi } from '../services/api';

export const useMonitors = (userToken: string | null, on401: () => void, refreshUserInfo: () => Promise<void>) => {
  const [monitors, setMonitors] = useState<MonitorTask[]>([]);
  const [loading, setLoading] = useState(false);
  const [refreshing, setRefreshing] = useState(false);
  const [trashMonitors, setTrashMonitors] = useState<MonitorTask[]>([]);
  const [trashLoading, setTrashLoading] = useState(false);

  // 获取监控任务列表
  const fetchMonitors = useCallback(async () => {
    setLoading(true);
    try {
      const data = await monitorApi.getMonitors(userToken, on401);
      setMonitors(data);
    } catch (err) {
      console.error('获取监控列表失败:', err);
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, [userToken, on401]);

  // 获取回收站列表
  const fetchTrashMonitors = useCallback(async () => {
    setTrashLoading(true);
    try {
      const data = await monitorApi.getTrash(userToken, on401);
      setTrashMonitors(data);
    } catch (err) {
      console.error('获取回收站列表失败:', err);
    } finally {
      setTrashLoading(false);
    }
  }, [userToken, on401]);

  // Phase 2: 创建监控任务（费用确认已在 CreateMonitorModal 中完成）
  const createMonitor = useCallback(async (
    params: MonitorCreatePayload,
    searchDays: number,
    onSuccess?: () => void
  ): Promise<boolean> => {
    try {
      const requestPayload: MonitorCreatePayload = {
        ...params,
        // 保证搜索链路传来的真实天数不会在创建监控时丢失
        days: params.days ?? searchDays,
      };

      // 直接调用 API 创建任务（后端会进行实际扣费和余额校验）
      await monitorApi.createMonitor(userToken, requestPayload, on401);
      await refreshUserInfo();
      await fetchMonitors();
      if (onSuccess) onSuccess();
      return true;
    } catch (err: unknown) {
      const errMsg = err instanceof Error ? err.message : 'Unknown error';
      // Phase 2: 检测余额不足错误
      if (errMsg.includes('Insufficient credits') || errMsg.includes('402')) {
        Alert.alert(
          '💎 Insufficient Credits',
          'You don\'t have enough credits. Please top up first.',
          [{ text: 'OK', style: 'cancel' }]
        );
      } else {
        Alert.alert('Failed', errMsg);
      }
      return false;
    }
  }, [userToken, on401, refreshUserInfo, fetchMonitors]);
