export const handle401 = async (): Promise<void> => {
  try {
    await storage.removeItem(TOKEN_KEY);
    await storage.removeItem(USER_KEY);
  } catch (e) {
    // 即使 storage 操作失败也不阻塞流程
    console.warn('[handle401] Failed to clear storage:', e);
  }
};
