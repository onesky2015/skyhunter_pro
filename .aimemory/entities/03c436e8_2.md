export const apiRequest = async (
  endpoint: string,
  options: RequestInit = {},
  token: string | null,
  on401?: () => void
): Promise<Response> => {
  const headers = getAuthHeaders(token, options.headers as Record<string, string>);
  const response = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers,
  });
  
  if (response.status === 401) {
    // 无论是否提供 on401 回调，都先严格清除本地存储的 token
    await handle401();
    // 再触发回调（通常用于更新 React 状态并重定向到登录页）
    if (on401) on401();
  }
  
  return response;
};
