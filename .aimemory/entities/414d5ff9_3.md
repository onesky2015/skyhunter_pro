```typescript
import React from 'react';
import {
  View,
  Text,
  Modal,
  ScrollView,
  TouchableOpacity,
  StyleSheet,
  ActivityIndicator,
} from 'react-native';
import { MonitorTask } from '../types';
import { formatPrice } from '../constants/utils';

interface TrashModalProps {
  visible: boolean;
  onClose: () => void;
  monitors: MonitorTask[];
  loading: boolean;
  onRestore: (id: number) => void;
}

export const TrashModal: React.FC<TrashModalProps> = ({
  visible,
  onClose,
  monitors,
  loading,
  onRestore,
}) => {
  return (
    <Modal
      visible={visible}
      transparent
      animationType="slide"
      onRequestClose={onClose}
    >
      <View style={styles.overlay}>
        <View style={styles.modal}>
          <View style={styles.header}>
            <Text style={styles.title}>🗑️ TRASH BIN</Text>
            <TouchableOpacity onPress={onClose}>
              <Text style={styles.closeButton}>✕</Text>
            </TouchableOpacity>
          </View>

          {loading ? (
            <View style={styles.center}>
              <ActivityIndicator size="large" color="#00FF00" />
            </View>
          ) : monitors.length === 0 ? (
            <View style={styles.center}>
              <Text style={styles.emptyText}>Trash is empty 🎉</Text>
            </View>
          ) : (
            <ScrollView style={styles.list}>
              {monitors.map((task) => (
                <View key={task.id} style={styles.item}>
                  <View style={styles.itemContent}>
                    <Text style={styles.itemRoute}>
                      {task.origin} → {task.destination}
                    </Text>
                    <Text style={styles.itemMeta}>
                      {task.start_date} · {task.days}d
                    </Text>
                    {task.current_price != null && (
                      <Text style={styles.itemPrice}>
                        {formatPrice(task.current_price)}
                      </Text>
                    )}
                  </View>
                  <TouchableOpacity
                    style={styles.restoreButton}
                    onPress={() => onRestore(task.id)}
                  >
                    <Text style={styles.restoreText}>♻️ Restore</Text>
                  </TouchableOpacity>
                </View>
              ))}
            </ScrollView>
          )}

          <TouchableOpacity style={styles.closeButtonBottom} onPress={onClose}>
            <Text style={styles.closeButtonText}>CLOSE</Text>
          </TouchableOpacity>
        </View>
      </View>
    </Modal>
  );
};

const styles = StyleSheet.create({
  overlay: {
    flex: 1,
    backgroundColor: 'rgba(0, 0, 0, 0.8)',
    justifyContent: 'flex-end',
  },
  modal: {
    backgroundColor: '#111111',
    borderTopLeftRadius: 16,
    borderTopRightRadius: 16,
    padding: 20,
```
