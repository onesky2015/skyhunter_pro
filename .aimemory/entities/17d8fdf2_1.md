// filename: constants/api.ts
import { Platform } from 'react-native';
import Constants from 'expo-constants';

// 从 app.json extra 配置读取生产环境 API URL
const PRODUCTION_API_URL = Constants.expoConfig?.extra?.apiUrl || 'https://api.skyhunter.app';

export const getApiUrl = (): string => {
  if (!__DEV__) {
    // 生产环境：始终使用配置的生产 API URL
    return PRODUCTION_API_URL;
  }
  // 开发环境
  if (Platform.OS === 'android') {
    return 'http://10.0.2.2:8000';
  }
  if (Platform.OS === 'web') {
    return 'http://localhost:8000';
  }
  return 'http://localhost:8000'; // iOS 模拟器
};

export const API_URL = getApiUrl();

// Token 存储键
export const TOKEN_KEY = 'skyhunter_token';
export const USER_KEY = 'skyhunter_user';
